# Go packages installed with `go install`, into ~/.local/bin (GOBIN).
#
# Format:  module@version[:binary]
# Module paths are stable URLs; the GitHub release tarballs are not — sesh's
# asset was renamed amd64 -> x86_64 and silently 404'd every install.
#
# sesh's go.mod asks for a newer Go than Ubuntu ships. GOTOOLCHAIN=auto (the
# default since 1.21) downloads the right one on first build.

github.com/joshmedeski/sesh/v2@latest:sesh   # tmux session manager
# lazygit's own docs steer people away from `go install` (the resulting binary
# has no embedded version info), but it's otherwise a normal `go install`.
github.com/jesseduffield/lazygit@latest:lazygit  # git TUI

# --- apt fallbacks ---------------------------------------------------------
# As in packages-cargo.txt: only ever built on a box where apt didn't supply
# them. gojq stands in for jq — install_shims symlinks it as `jq` if jq is
# missing entirely.
github.com/junegunn/fzf@latest:fzf           # fuzzy finder
github.com/itchyny/gojq/cmd/gojq@latest:gojq # jq, in Go (near-drop-in)
